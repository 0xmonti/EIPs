#include <iostream>
#include <string>
#include <sstream>

class MontiStringImmortal {
private:
    static constexpr const char* IDENTITY = "JOHNCHARLESMONTI.COM";
    static constexpr const char* ANSI_RESET = "\033[0m";
    static constexpr const char* ANSI_CYAN  = "\033[38;2;0;255;234m";
    static constexpr const char* ANSI_GREEN = "\033[38;2;63;185;80m";
    static constexpr const char* ANSI_WARN  = "\033[38;2;255;165;0m";

public:
    static std::string format_monti_string(const std::string& src_addr, 
                                           const std::string& dst_addr, 
                                           uint64_t amount_eth) {
        std::ostringstream ansi_stream;
        
        ansi_stream << ANSI_CYAN  << "[" << IDENTITY << "::MONTI_STRING_ANSI_ENGINE]" << ANSI_RESET << "\n"
                    << ANSI_WARN  << "-> DEDUCT: " << src_addr << " (-" << amount_eth << " ETH)\n" << ANSI_RESET
                    << ANSI_GREEN << "-> CREDIT: " << dst_addr << " (+" << amount_eth << " ETH)\n" << ANSI_RESET
                    << ANSI_CYAN  << "[STATUS: STATE_UNIFORMITY_VERIFIED]" << ANSI_RESET;

        return ansi_stream.str();
    }
};

int main() {
    std::string output = MontiStringImmortal::format_monti_string(
        "0xAb5801a7D398351b8bE11C439e05C5B3259aeC9B",
        "0x15E55EF43efA8348dDaeAa455F16C43B64917e3c",
        100000
    );
    std::cout << output << std::endl;
    return 0;
}
